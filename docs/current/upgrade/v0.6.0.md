# 升级到 v0.6.0

## 概述

本指南适用于从 v0.5.10 升级到 v0.6.0。如果您使用 SQLite，则无需升级数据库。如果您使用 MySQL，则需要升级数据库。

## 准备工作

### 备份数据库

为防止数据丢失，建议在升级前备份数据库。备份方式根据您的数据库类型而定。

## 升级

### 停止 DB-GPT 服务

根据您的启动方式停止 DB-GPT 服务。

### 升级数据库
执行以下 SQL 来升级数据库。

:::tip
版本升级所需的资产文件放在[此处](https://github.com/eosphoros-ai/DB-GPT/tree/main/assets/schema/upgrade)。
:::

```sql
USE dbgpt;
-- chat_history
ALTER TABLE chat_history ADD COLUMN `app_code` varchar(255) DEFAULT NULL COMMENT '应用唯一编码' after `message_ids`;

-- gpts_app
ALTER TABLE gpts_app ADD COLUMN `published` varchar(64) DEFAULT 'false' COMMENT '是否已发布？';
ALTER TABLE gpts_app ADD COLUMN `param_need` text DEFAULT NULL COMMENT '应用支持的参数信息';
ALTER TABLE gpts_app ADD COLUMN `admins` text DEFAULT NULL COMMENT '管理员';

-- connect_config
ALTER TABLE connect_config ADD COLUMN `user_name` varchar(255) DEFAULT NULL COMMENT '用户名';
ALTER TABLE connect_config ADD COLUMN `user_id` varchar(255) DEFAULT NULL COMMENT '用户 id';

-- document_chunk
ALTER TABLE document_chunk ADD COLUMN `questions` text DEFAULT NULL COMMENT '分块相关问题';

-- knowledge_document
ALTER TABLE knowledge_document ADD COLUMN `doc_token` varchar(100) DEFAULT NULL COMMENT '文档 token';
ALTER TABLE knowledge_document ADD COLUMN `questions` text DEFAULT NULL COMMENT '文档相关问题';

-- gpts_messages
ALTER TABLE gpts_messages ADD COLUMN `is_success` int(4) NULL DEFAULT 0 COMMENT 'Agent 消息是否成功';
ALTER TABLE gpts_messages ADD COLUMN `app_code` varchar(255) NOT NULL COMMENT '当前 AI 助手编码';
ALTER TABLE gpts_messages ADD COLUMN `app_name` varchar(255) NOT NULL COMMENT '当前 AI 助手名称';
ALTER TABLE gpts_messages ADD COLUMN `resource_info` text DEFAULT NULL COMMENT '当前会话资源信息';

-- prompt_manage
ALTER TABLE prompt_manage ADD COLUMN `prompt_code` varchar(255) NULL COMMENT 'Prompt 编码';
ALTER TABLE prompt_manage ADD COLUMN `response_schema` text NULL COMMENT 'Prompt 响应模式';
ALTER TABLE prompt_manage ADD COLUMN `user_code` varchar(128) NULL COMMENT '用户编码';

-- chat_feed_back
ALTER TABLE chat_feed_back ADD COLUMN `message_id` varchar(255) NULL COMMENT '消息 id';
ALTER TABLE chat_feed_back ADD COLUMN `feedback_type` varchar(50) NULL COMMENT '反馈类型，喜欢或不喜欢';
ALTER TABLE chat_feed_back ADD COLUMN `reason_types` varchar(255) NULL COMMENT '反馈原因分类';
ALTER TABLE chat_feed_back ADD COLUMN `user_code` varchar(128) NULL COMMENT '用户编码';
ALTER TABLE chat_feed_back ADD COLUMN `remark` text NULL COMMENT '反馈备注';

-- dbgpt_serve_flow
ALTER TABLE dbgpt_serve_flow ADD COLUMN `variables` text DEFAULT NULL COMMENT '流程变量，JSON 格式';

-- dbgpt.recommend_question 定义
CREATE TABLE `recommend_question` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT COMMENT '自增 id',
  `gmt_create` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `gmt_modified` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `app_code` varchar(255) DEFAULT NULL COMMENT '当前 AI 助手编码',
  `question` text DEFAULT NULL COMMENT '问题',
  `user_code` int(11) DEFAULT NULL COMMENT '用户编码',
  `sys_code` varchar(255) DEFAULT NULL COMMENT '系统应用编码',
  `valid` varchar(10) DEFAULT 'true' COMMENT '是否有效，true/false',
  `chat_mode` varchar(255) DEFAULT NULL COMMENT '对话场景模式，chat_knowledge...',
  `params` text DEFAULT NULL COMMENT '问题参数',
  `is_hot_question` varchar(10) DEFAULT 'false' COMMENT '是否为热门推荐问题？',
  PRIMARY KEY (`id`),
  KEY `idx_rec_q_app_code` (`app_code`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT="AI 应用相关推荐问题";

-- dbgpt.user_recent_apps 定义
CREATE TABLE `user_recent_apps` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT COMMENT '自增 id',
  `gmt_create` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `gmt_modified` timestamp NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '最后更新时间',
  `app_code` varchar(255) DEFAULT NULL COMMENT 'AI 助手编码',
  `last_accessed` timestamp NULL DEFAULT NULL COMMENT '用户最近使用时间',
  `user_code` varchar(255) DEFAULT NULL COMMENT '用户编码',
  `sys_code` varchar(255) DEFAULT NULL COMMENT '系统应用编码',
  PRIMARY KEY (`id`),
  KEY `idx_user_r_app_code` (`app_code`),
  KEY `idx_last_accessed` (`last_accessed`),
  KEY `idx_user_code` (`user_code`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户最近使用的应用';

-- dbgpt.dbgpt_serve_file 定义
CREATE TABLE `dbgpt_serve_file` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT '自增 id',
  `bucket` varchar(255) NOT NULL COMMENT '桶名称',
  `file_id` varchar(255) NOT NULL COMMENT '文件 id',
  `file_name` varchar(256) NOT NULL COMMENT '文件名',
  `file_size` int DEFAULT NULL COMMENT '文件大小',

  `storage_path` varchar(512) NOT NULL COMMENT '存储路径',
  `uri` varchar(512) NOT NULL COMMENT '文件 URI',
  `custom_metadata` text DEFAULT NULL COMMENT '自定义元数据，JSON 格式',
  `file_hash` varchar(128) DEFAULT NULL COMMENT '文件哈希',
  `user_name` varchar(128) DEFAULT NULL COMMENT '用户名',
  `sys_code` varchar(128) DEFAULT NULL COMMENT '系统编码',
  `gmt_created` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
  `gmt_modified` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录更新时间',
  PRIMARY KEY (`id`),

) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- dbgpt.dbgpt_serve_variables 定义
CREATE TABLE `dbgpt_serve_variables` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT '自增 id',
  `key` varchar(128) NOT NULL COMMENT '变量键',
  `name` varchar(128) DEFAULT NULL COMMENT '变量名称',
  `label` varchar(128) DEFAULT NULL COMMENT '变量标签',
  `value` text DEFAULT NULL COMMENT '变量值，JSON 格式',
  `value_type` varchar(32) DEFAULT NULL COMMENT '变量值类型（string, int, float, bool）',
  `category` varchar(32) DEFAULT 'common' COMMENT '变量类别（common 或 secret）',
  `encryption_method` varchar(32) DEFAULT NULL COMMENT '变量加密方法（fernet, simple, rsa, aes）',
  `salt` varchar(128) DEFAULT NULL COMMENT '变量盐值',
  `scope` varchar(32) DEFAULT 'global' COMMENT '变量作用域（global, flow, app, agent, datasource, flow_priv, agent_priv 等）',
  `scope_key` varchar(256) DEFAULT NULL COMMENT '变量作用域键，默认为空，当 scope 为 "flow_priv" 时，scope_key 为流程的 dag id',
  `enabled` int DEFAULT 1 COMMENT '变量是否启用，0：禁用，1：启用',
  `description` text DEFAULT NULL COMMENT '变量描述',
  `user_name` varchar(128) DEFAULT NULL COMMENT '用户名',
  `sys_code` varchar(128) DEFAULT NULL COMMENT '系统编码',
  `gmt_created` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '记录创建时间',
  `gmt_modified` datetime DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '记录更新时间',
  PRIMARY KEY (`id`),
  KEY `ix_your_table_name_key` (`key`),
  KEY `ix_your_table_name_name` (`name`)
) ENGINE=InnoDB AUTO_INCREMENT=1 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- dbgpt.dbgpt_serve_dbgpts_my 定义
CREATE TABLE `dbgpt_serve_dbgpts_my` (

  `name` varchar(255) NOT NULL COMMENT '插件名称',
```
